package main

import (
	"os/exec"
	"testing"
)

func TestGokitPython(t *testing.T) {
	if _, err := exec.LookPath("uv"); err != nil {
		t.Skip("uv not installed")
	}
	w, err := gokitPython()
	if err != nil {
		t.Fatal(err)
	}
	defer w.Close()
	var resp struct{ Echo map[string]string }
	if err := w.Call(map[string]string{"hello": "world"}, &resp); err != nil {
		t.Fatal(err)
	}
	if resp.Echo["hello"] != "world" {
		t.Fatalf("unexpected response %+v", resp)
	}
	if err := w.Call(map[string]string{"fail": "on purpose"}, nil); err == nil {
		t.Fatal("expected worker error")
	}
}
